/* index.css - The Magic Paint and Style Book for Our Website! */
/* Think of this like the instruction manual for how everything should look! */

/* 🎪 STEP 1: Import Google Fonts (Like ordering special handwriting styles!) */
@import url('https://fonts.googleapis.com/css2?family=Poppins:wght@400;600;700&family=Open+Sans:wght@400;500&display=swap');
/* This says: "Dear Google, please send us the Poppins and Open Sans fonts!" */
/* Like ordering special art supplies from an art store! */

/* 🎨 STEP 2: Tailwind CSS Directives (The magic words that activate Tailwind!) */
@tailwind base;        /* 🏗️ "Build the foundation/base styles first!" */
@tailwind components;  /* 🧱 "Add all the pre-made building blocks next!" */
@tailwind utilities;   /* 🛠️ "Add all the little helper tools last!" */
/* These are like saying "Abracadabra!" to make Tailwind magic work! */

/* 🎯 STEP 3: Custom Base Styles (Setting up our website's basic look) */
/* @layer base means: "These styles go with the foundation/base layer" */
@layer base {
  /* 🧍 BODY: Style for the whole website body (like dressing the mannequin!) */
  body {
    @apply font-body text-gray-800;  /* "Use Open Sans font with dark gray text!" */
    /* font-body = Open Sans (from our tailwind.config.js) */
    /* text-gray-800 = A nice dark gray color (like charcoal!) */
  }
  
  /* 🎪 HEADINGS: Style for all heading tags (h1, h2, h3, etc.) */
  h1, h2, h3, h4, h5, h6 {
    @apply font-display font-bold;  /* "Use Poppins font and make it bold!" */
    /* font-display = Poppins (from our tailwind.config.js) */
    /* font-bold = Make it extra thick and strong! */
  }
}

/* 🛠️ STEP 4: Custom Components (Our own special building blocks!) */
/* @layer components means: "These are our custom-made components" */
@layer components {
  /* 🔘 PRIMARY BUTTON: Our main action button (like a big red "GO!" button!) */
  .btn-primary {
    @apply bg-nova-orange text-white font-bold px-6 py-3 rounded-full hover:bg-nova-red transition-colors duration-200;
    /* Let's break this down: */
    /* bg-nova-orange = Orange background (our special color!) */
    /* text-white = White text (easy to read on orange!) */
    /* font-bold = Thick, strong text */
    /* px-6 py-3 = Padding: 6 units left/right, 3 units top/bottom */
    /* rounded-full = Make it completely round (like a pill!) */
    /* hover:bg-nova-red = When mouse hovers, turn dark red! */
    /* transition-colors = Smooth color change (not sudden jump!) */
    /* duration-200 = Color change takes 200 milliseconds (blink of an eye!) */
  }
  
  /* 🔳 SECONDARY BUTTON: Our alternative button (like a "Maybe Later" button) */
  .btn-secondary {
    @apply border-2 border-nova-orange text-nova-orange font-bold px-6 py-3 rounded-full hover:bg-nova-orange hover:text-white transition-colors duration-200;
    /* border-2 border-nova-orange = Orange outline (not filled in!) */
    /* text-nova-orange = Orange text */
    /* hover:bg-nova-orange = When hovering, fill with orange! */
    /* hover:text-white = When hovering, text turns white! */
  }
  
  /* 🃏 CARD: Our information boxes (like trading cards or picture frames!) */
  .card {
    @apply bg-white rounded-xl shadow-lg hover:shadow-xl transition-shadow duration-300;
    /* bg-white = White background (clean and fresh!) */
    /* rounded-xl = Extra rounded corners (softer look!) */
    /* shadow-lg = Big shadow (makes it pop off the page!) */
    /* hover:shadow-xl = When hovering, shadow gets even bigger! */
    /* transition-shadow = Smooth shadow animation */
    /* duration-300 = Animation takes 300 milliseconds */
  }
}

/* 📚 Kid-Friendly Explanations (AFTER the @layer, not inside!) */

/* 🎨 PRIMARY BUTTON EXPLANATION:
   - Orange background (bg-nova-orange)
   - White text (text-white) 
   - Bold font (font-bold)
   - Padding: 6 left/right, 3 top/bottom (px-6 py-3)
   - Round shape (rounded-full)
   - Hover turns dark red (hover:bg-nova-red)
   - Smooth color change (transition-colors)
   - 200ms animation (duration-200) */

/* 🎨 SECONDARY BUTTON EXPLANATION:
   - Orange border (border-2 border-nova-orange)
   - Orange text (text-nova-orange)
   - Hover fills with orange (hover:bg-nova-orange)
   - Hover text turns white (hover:text-white) */

/* 🎨 CARD EXPLANATION:
   - White background (bg-white)
   - Rounded corners (rounded-xl)
   - Big shadow (shadow-lg)
   - Hover shadow bigger (hover:shadow-xl)
   - Smooth shadow animation (transition-shadow)
   - 300ms animation (duration-300) */

The Magic of @apply:

```css
@apply bg-nova-orange text-white
```

This means: "Take all these Tailwind classes and apply them here!"

Without @apply:

```html
<button class="bg-nova-orange text-white font-bold px-6 py-3 rounded-full">
```

With @apply:

```css
.btn-primary {
  @apply bg-nova-orange text-white font-bold px-6 py-3 rounded-full;
}
```

```html
<button class="btn-primary">
```

It's like making a shortcut! 🎯

"Then, create our special building blocks using those base rules!"

Utilities Layer Last:

```css
@tailwind utilities;
```

"Finally, add all the little helper tools we might need!"
